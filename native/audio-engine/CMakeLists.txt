cmake_minimum_required(VERSION 3.20)
project(LouderMeAudioEngine LANGUAGES CXX)
include(CTest)
add_library(louderme_audio_engine STATIC src/AudioEngine.cpp)
target_compile_features(louderme_audio_engine PUBLIC cxx_std_17)
target_include_directories(louderme_audio_engine PUBLIC include)
set_target_properties(louderme_audio_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(louderme_audio_c SHARED src/AudioEngineC.cpp)
target_compile_features(louderme_audio_c PUBLIC cxx_std_17)
target_include_directories(louderme_audio_c PUBLIC include)
target_link_libraries(louderme_audio_c PRIVATE louderme_audio_engine)
set_target_properties(louderme_audio_c PROPERTIES OUTPUT_NAME louderme_audio_c)
if(MSVC)
  target_compile_options(louderme_audio_engine PRIVATE /W4 /WX)
else()
  target_compile_options(louderme_audio_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(louderme_audio_engine_tests tests/AudioEngineTests.cpp)
  target_link_libraries(louderme_audio_engine_tests PRIVATE louderme_audio_engine)
  add_test(NAME lou_own_dsp COMMAND louderme_audio_engine_tests)
  add_executable(louderme_audio_abi_tests tests/AudioEngineCAbiTests.cpp)
  target_include_directories(louderme_audio_abi_tests PRIVATE include)
  target_link_libraries(louderme_audio_abi_tests PRIVATE louderme_audio_c)
  add_test(NAME lou_abi_bridge COMMAND louderme_audio_abi_tests)
endif()
